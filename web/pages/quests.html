クエスト一覧

クエストとジャンクションの完了状況です。前提が完了していないものはチェックできません(この PC に保存)。

前提の札を押すと、その行へ移ります。「導線」には、完了までに必要な未完了のものを前提から順に並べています。MR(マスタリーランク)とそのほかの条件は案内だけで、チェックには使いません。※ の付いた行は、前提がまだ確かめきれていないものです。

/* ---------- クエスト一覧 ---------- */ /* 前提(requires)は API がそろえた形: node {id} / any {of: [id]} / mr {n} / note {text}。チェックできるかは state(locked は付けられない)。 付ける・外すの最終判断はサーバー(POST /api/quests が 409 で断る)。状態は前提の完了で変わるので、返ってきた一覧ごと置き換える */ let quests = null, qsCat = ""; const qsArming = createArming(() => renderQuests()); // 導線をまとめて付ける2段階の確認(parts.html) const qsOpen = new Set(); // 導線を開いている行の id(描き直しても開いたままにする) const qsLabel = r => r.kind === "junction" ? fmt("quest.junction", { to: tName("planet.", r.to) }) : nameByLang(r); function qsFilter() { const q = $("#qs-q").value.trim().toLowerCase(), st = $("#qs-state").value; return quests.rows.filter(r => (!qsCat || r.cat === qsCat) && (!st || (st === "todo" ? !r.done : r.state === st)) && (!q || [r.en, r.ja, r.to, qsLabel(r)].some(x => (x || "").toLowerCase().includes(q)))); } function qsReqs(r, by) { const chip = id => by[id] ? `` : ""; const parts = r.requires.map(q => q.type === "node" ? chip(q.id) : q.type === "any" ? `${esc(t("quest.req.any"))}${q.of.map(chip).join("")}` : q.type === "mr" ? `${esc(fmt("quest.req.mr", { n: q.n }))}` : `${esc(renderTxt(q.text))}`); let html = `
${parts.join("") || `${esc(t("quest.req.none"))}`}
`; if (r.route.length) { const n = r.route.length; html += `
${esc(fmt("quest.route", { n }))}
    ${r.route.map(id => `
  1. ${by[id] ? esc(qsLabel(by[id])) : ""}
    2. `).join("")}
`; } return html; } function renderQuests() { if (!quests) return; const by = Object.fromEntries(quests.rows.map(r => [r.id, r])), sm = quests.summary; $("#qs-cats").innerHTML = [["", t("src.all"), sm.all], ...quests.cats.filter(k => sm[k] && sm[k].total).map(k => [k, tName("quest.cat.", k), sm[k]])] .map(([k, v, o]) => ``).join(""); const list = qsFilter(), cur = (qsCat ? sm[qsCat] : sm.all) || sm.all, pctDone = ratePct(cur.done, cur.total); $("#qs-summary").innerHTML = `${esc(qsCat ? tName("quest.cat.", qsCat) : t("quest.all"))} ${esc(fmt("quest.summary", { done: cur.done, total: cur.total, pct: pctDone.toFixed(1) }))} ${progressBar(pctDone)} ${esc(fmt("quest.summary.ready", { n: cur.ready }))}${esc(fmt("common.shown", { n: list.length }))} ${quests.export ? "" : `${esc(t("quest.noexport"))}`}`; const mark = r => !r.flow ? `※` : !r.verified ? `※` : ""; drawTable($("#qs-out"), list.length ? `
${list.map(r => ``).join("")}
${esc(t("quest.col.done"))}${esc(t("quest.col.name"))}${esc(t("quest.col.cat"))}${esc(t("quest.col.state"))}${esc(t("quest.col.req"))}
${esc(qsLabel(r))}${mark(r)}${r.kind === "junction" ? `${esc(fmt("quest.junction.from", { from: tName("planet.", r.from) }))}` : subEn(r)} ${esc(tName("quest.cat.", r.cat))} ${esc(tName("quest.state.", r.state))}${r.done && r.missing.length ? `!` : ""} ${qsReqs(r, by)}
` : noData(t("common.none")), [qsCat, $("#qs-q").value, $("#qs-state").value]); } async function setQuests(ids, done) { $("#qs-err").textContent = ""; try { const res = await api("/api/quests", { method: "POST", body: { ids, done } }); quests.rows = res.rows; quests.summary = res.summary; } catch (e) { $("#qs-err").textContent = e.message; } renderQuests(); // 断られたときもチェックの見た目を元に戻す } function qsGoto(id) { let tr = $(`#qs-out tr[data-id="${CSS.escape(id)}"]`); if (!tr) { qsCat = ""; $("#qs-q").value = ""; $("#qs-state").value = ""; renderQuests(); tr = $(`#qs-out tr[data-id="${CSS.escape(id)}"]`); } if (!tr) return; tr.scrollIntoView({ block: "center", behavior: "smooth" }); tr.classList.add("qs-flash"); setTimeout(() => tr.classList.remove("qs-flash"), 1600); } $("#qs-cats").addEventListener("click", e => { const c = e.target.closest("[data-qcat]"); if (!c) return; qsCat = c.dataset.qcat; qsArming.clear(); $("#qs-err").textContent = ""; renderQuests(); }); $("#qs-out").addEventListener("change", e => { const cb = e.target.closest("input.own"); if (cb) setQuests([cb.closest("tr").dataset.id], cb.checked); }); $("#qs-out").addEventListener("click", e => { const go = e.target.closest("[data-goto]"); if (go) { qsGoto(go.dataset.goto); return; } const b = e.target.closest("[data-route]"); if (!b || !quests) return; const id = b.dataset.route, r = quests.rows.find(x => x.id === id); if (!r) return; if (!qsArming.press(id)) return; // 1回目は確認の見た目にするだけ qsOpen.delete(id); setQuests([...r.route, id], true); // 前提から順に。サーバーは一緒に付けるものも前提として数える }); // 導線の開け閉めを覚える(toggle は泡立たないので捕捉で受ける) $("#qs-out").addEventListener("toggle", e => { const d = e.target.closest && e.target.closest("details.qs-route"); if (d) (d.open ? qsOpen.add(d.dataset.id) : qsOpen.delete(d.dataset.id)); }, true); $("#qs-q").addEventListener("input", debounce(() => { qsArming.clear(); $("#qs-err").textContent = ""; renderQuests(); }, 150)); $("#qs-state").addEventListener("change", () => { qsArming.clear(); $("#qs-err").textContent = ""; renderQuests(); }); loaders.quests = async () => { applyViewText("quests"); if (quests) { renderQuests(); return; } $("#qs-out").innerHTML = `
${esc(t("common.loading"))}
`; try { quests = await api("/api/quests"); renderQuests(); } catch (e) { $("#qs-cats").innerHTML = ""; $("#qs-summary").innerHTML = ""; $("#qs-out").innerHTML = noData(e.message); } }; relabel.quests = () => { applyViewText("quests"); $("#qs-err").textContent = ""; renderQuests(); }; // エラー文は前の言語のままになるので消す dataResets.quests = () => { quests = null; };
