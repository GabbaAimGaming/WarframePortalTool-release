アルケイン所持

公式データ(ExportRelicArcane.json)のアルケインを装備先ごとに一覧にします。左端のチェックで所持として記録し、ランクを選べます(この PC に保存)。入手先(ドロップ表・シンジケート・商人など)は MOD 入手一覧と同じ形で出し、行をクリックすると詳細を開きます。名前とレア度はカタカナで表示します(英語名でも絞り込めます)。

種類(装備先)は名前から判定しています(Arcane・Molt・Theorem は Warframe、Magus・Emergence はオペレーター、Virtuos・Eternal はアンプ、Pax・Residual は Kitgun、Exodia は Zaw など)。判定できない新しいアルケインは「その他」に入ります。テクトリストアーティファクト用の Zid-An は、公式データにまだ無いためドロップ表から補います(英語名・レア度なし・最大ランク 5。公式データに入ればそちらを使います)。公式データにあってもゲームに実装されていない 7 件(Arcane Survival・Shield・Defense・Detoxifier・Liquid・Protection・Temperance)は載せません。チェックを外すとランクも消えます。新しいアルケインはゲームのアップデートで増えるので、「データ更新」で取り直してください。

/* ---------- アルケイン所持 ---------- */ let arcanes = null, acCat = "warframe", acOpen = null; const acSort = { key: "name", dir: "asc" }; const acGetters = { owned: r => (r.owned ? 1 : 0), name: r => nameByLang(r).toLowerCase(), rarity: r => rarityRank(r.rarity), rank: r => (r.owned ? r.rank : -1), total: r => r.total, best: r => bestRank(r.best), }; /* 名前: 日本語の画面は日本語名(無ければ英名)、ほかは英名 */ function acFilter(inCat) { const q = $("#ac-q").value.trim().toLowerCase(), st = $("#ac-state").value, src = $("#ac-src").value; return inCat.filter(r => (!q || [r.en, r.ja].some(x => (x || "").toLowerCase().includes(q))) && (!src || (src === "any" ? r.total > 0 : src === "none" ? r.total === 0 : (r.counts[src] || 0) > 0)) && (!st || (st === "owned" && r.owned) || (st === "unowned" && !r.owned) || (st === "maxed" && r.owned && r.rank === r.max_rank) || (st === "notmax" && r.owned && r.rank < r.max_rank))); } function acSummary() { const sm = { all: { total: 0, owned: 0, maxed: 0 } }; arcanes.cats.forEach(c => sm[c] = { total: 0, owned: 0, maxed: 0 }); arcanes.rows.forEach(r => [sm[r.cat], sm.all].forEach(o => { if (!o) return; o.total++; if (r.owned) { o.owned++; if (r.rank === r.max_rank) o.maxed++; } })); return sm; } const acRankHtml = r => r.owned ? `/ ${r.max_rank}` : `-/ ${r.max_rank}`; /* 入手先(MOD 入手一覧と同じ形): 件数のバッジと、一番よい入手先(確率が無ければ詳細 = ランクと名声・値段などを小さく) */ const acCounts = r => r.total ? `

${th(esc(t("mods.col.own")), "owned", acSort, "chk")}${th(esc(t("ms.col.name")), "name", acSort)}${th(esc(t("src.col.rarity")), "rarity", acSort)}${th(esc(t("ac.col.rank")), "rank", acSort, "num")}${th(esc(t("mods.col.total")), "total", acSort)}${th(esc(t("mods.col.best")), "best", acSort)}${list.map(r => ` ${acOpen === r.en ? `` : ""}`).join("")}

${acDetail.html(r.en)}
