エリアから探す

惑星またはノード名を選ぶと、出現ステージごとに素材と確率を表示します。

/* ---------- エリアから探す ---------- */ let areaSel = null, areaList = null, areaRes = null; // 候補・結果は言語の切り替えで描き直すため手元に残す function renderAreaChips() { const list = areaList || []; $("#area-chips").innerHTML = list.length ? list.map(i => ``).join("") : `${esc(t("area.nomatch"))}`; } // 選んでいる種類(AREA_KINDS のキー。初期は AREA_KIND_DEFAULT。覚えない) const areaKinds = kindChecks("area", CONST.AREA_KIND_DEFAULT); async function loadAreaChips() { try { areaList = await api(`/api/suggest/area?q=${encodeURIComponent($("#area-q").value)}`); renderAreaChips(); return areaList; } catch (e) { areaList = null; $("#area-out").innerHTML = noData(e.message); return []; } } function renderAreaResult() { const r = areaRes; if (!areaSel || !r) { if (!areaSel) $("#area-out").innerHTML = `
${esc(t("area.pick"))}
`; return; } const title = esc(renderTxt(r.title)); if (!r.nodes.length) { $("#area-out").innerHTML = `

${title}

` + noData(t("area.none")); return; } const head = ["stage", "item", "chance", "rarity"].map(c => `${esc(t("area.col." + c))}`).join(""); $("#area-out").innerHTML = `

${title}

${esc(fmt("area.count", { n: r.nodes.length }))}
${r.nodes.map(n => { const max = Math.max(1, ...n.stages.flatMap(s => s.items.map(i => i.chance))); const rows = n.stages.flatMap(s => s.items.map((i, k) => `${k === 0 ? esc(renderTxt(s.stage)) : ""}${esc(nameByLang(i))}${esc(t("area.kind." + i.kind))}
${pct(i.chance)}
${rarity(i.rarity)}`)).join(""); return `
${esc(n.node)}${esc(renderTxt(n.planet))} / ${esc(renderTxt(n.mode))}${n.event ? `${esc(t("src.event"))}` : ""}
${rows ? `
${head}${rows}
` : `
${esc(t("area.nomaterial"))}
`}
`; }).join("")}
`; } async function loadAreaResult() { areaRes = null; if (!areaSel) { renderAreaResult(); return; } const kinds = areaKinds(); if (!kinds.length) { areaRes = null; $("#area-out").innerHTML = `
${esc(t("area.nokind"))}
`; return; } try { areaRes = await api(`/api/area?type=${areaSel.type}&key=${encodeURIComponent(areaSel.key)}&kinds=${kinds.join(",")}&events=${$("#area-ev").checked ? 1 : 0}`); renderAreaResult(); } catch (e) { $("#area-out").innerHTML = noData(e.message); } } $("#area-q").addEventListener("input", debounce(async () => { const list = await loadAreaChips(); const q = $("#area-q").value.trim(); if (q && list && list.length === 1) { areaSel = { type: list[0].type, key: list[0].key }; await loadAreaChips(); loadAreaResult(); } })); $("#area-kinds").addEventListener("change", e => { if (e.target.dataset.kind) loadAreaResult(); }); $("#area-ev").addEventListener("change", loadAreaResult); $("#area-chips").addEventListener("click", e => { const c = e.target.closest(".chip"); if (!c) return; areaSel = { type: c.dataset.type, key: c.dataset.key }; $("#area-chips").querySelectorAll(".chip").forEach(x => x.setAttribute("aria-pressed", x === c)); loadAreaResult(); }); loaders.area = async () => { applyViewText("area"); const s = await refreshStatus(); if (!s || !s.has_drops) { $("#area-chips").innerHTML = ""; $("#area-out").innerHTML = noData(t("area.nodata")); return; } await loadAreaChips(); await loadAreaResult(); }; relabel.area = () => { applyViewText("area"); if (areaList) renderAreaChips(); renderAreaResult(); };
