MOD 効果検索

影響先(例: 威力・効率)を入力すると、その数値を変える MOD をランク0〜最大の効果つきで一覧にします。候補を選ばずに Enter で、効果テキストの全文検索になります。

/* ---------- MOD 効果検索 ---------- */ /* 入力に部分一致する影響先(data/mod_stats/targets.json の表示名)を候補に出し、選んだ影響先の MOD を /api/modstats で一覧にする。 1つに決まれば(完全一致・候補が1件)すぐ探し、候補が無ければ効果テキストの全文検索。Enter か「全文で探す」でも全文検索。 区分のチェックは localStorage(wf-modstats-groups)に覚える(覚えていなければ全部)。効果テキストは画面の言語(ja が無ければ en) */ let mfxCands = [], mfxPick = null, mfxRes = null, mfxMsg = "modstats.empty"; // 候補・選んだもの({target} か {text})・結果・結果が無いときの案内 const mfxBoxes = () => [...document.querySelectorAll("#mfx-groups input[data-group]")]; const mfxGroups = () => mfxBoxes().filter(x => x.checked).map(x => x.dataset.group); (() => { const saved = pref.json("wf-modstats-groups"); if (Array.isArray(saved)) mfxBoxes().forEach(x => { x.checked = saved.includes(x.dataset.group); }); })(); const mfxNote = key => { mfxRes = null; mfxMsg = key; $("#mfx-out").innerHTML = `
${esc(t(key))}
`; }; function renderMfxChips() { const q = $("#mfx-q").value.trim(); $("#mfx-chips").innerHTML = mfxCands.map(c => ``).join("") + (q ? `` : ""); } function mfxCard(row, target, names) { const sub = lang === "ja" && row.ja && row.ja !== row.en ? row.en : ""; const compat = row.compat && row.compat !== row.compat.toUpperCase() ? `${esc(fmt("modstats.compat", { name: hasText("mod.compat." + row.compat) ? t("mod.compat." + row.compat) : row.compat }))}` : ""; const head = `
${esc(nameByLang(row))}${sub ? `${esc(sub)}` : ""}${ row.groups.map(g => `${esc(t("mod.group." + g))}`).join("")}${compat}
`; const ranks = lang === "ja" && row.stats_ja.length ? row.stats_ja : row.stats_en; if (!ranks.length) return `
${head}

${esc(nameByLang(row, "desc_ja", "desc_en"))}

`; // 列の見出し = その行の影響先(最大ランクの en で判定)。選んだ影響先の列は強調、条件付きは印を付ける const cols = row.lines.map(ln => ({ hit: !!target && ln.targets.includes(target), cond: ln.cond, label: ln.targets.map(k => (names[k] ? nameByLang(names[k]) : k)).join(" / ") || "-" })); const cls = j => (cols[j] && cols[j].hit ? ` class="hit"` : ""); return `
${head}
${cols.map((c, j) => `${esc(c.label)}${c.cond ? `${esc(t("modstats.cond"))}` : ""}`).join("")}${ranks.map((lns, i) => `${lns.map((s, j) => `${esc(s)}`).join("")}`).join("")}
${esc(t("modstats.rank"))}
${i}
`; } function renderMfxResult() { const r = mfxRes; if (!r) return; const title = r.target ? fmt("modstats.target", { name: nameByLang(r.target) }) : fmt("modstats.text", { q: r.query }); const key = r.target && r.target.key; $("#mfx-out").innerHTML = `

${esc(title)}

${esc(fmt("modstats.count", { n: r.total }))}${ r.updated ? `${esc(fmt("modstats.updated", { at: r.updated }))}` : ""}
${r.total > r.rows.length ? `

${esc(fmt("modstats.more", { n: r.rows.length }))}

` : ""} ${r.rows.length ? r.rows.map(row => mfxCard(row, key, r.names)).join("") : `
${esc(t("modstats.nomatch"))}
`}`; } async function mfxFetch() { if (!mfxPick) return; const groups = mfxGroups(); if (!groups.length) { mfxNote("modstats.nogroup"); return; } const pick = mfxPick, qs = pick.target ? `target=${encodeURIComponent(pick.target)}` : `q=${encodeURIComponent(pick.text)}`; try { const res = await api(`/api/modstats?${qs}&groups=${groups.join(",")}`); if (pick !== mfxPick) return; // 待っている間に選び直していたら捨てる mfxRes = { ...res, query: pick.text || "" }; renderMfxResult(); } catch (e) { mfxRes = null; $("#mfx-out").innerHTML = noData(e.message); } } async function mfxSearch() { const q = $("#mfx-q").value.trim(); mfxPick = null; if (!q) { mfxCands = []; renderMfxChips(); mfxNote("modstats.empty"); return; } let cands = []; try { cands = (await api(`/api/modstats/suggest?q=${encodeURIComponent(q)}`)).items; } catch (e) {} if ($("#mfx-q").value.trim() !== q) return; // 入力が変わっていたら捨てる mfxCands = cands; const exact = cands.find(c => [c.ja, c.en].some(n => n.toLowerCase() === q.toLowerCase())); if (exact || cands.length === 1) mfxPick = { target: (exact || cands[0]).key }; else if (!cands.length) mfxPick = { text: q }; renderMfxChips(); if (mfxPick) await mfxFetch(); else mfxNote("modstats.pick"); } function mfxPickText() { const q = $("#mfx-q").value.trim(); if (!q) return; mfxPick = { text: q }; renderMfxChips(); mfxFetch(); } $("#mfx-q").addEventListener("input", debounce(mfxSearch, 250)); $("#mfx-q").addEventListener("keydown", e => { if (e.key === "Enter" && !e.isComposing) { e.preventDefault(); mfxPickText(); } }); $("#mfx-chips").addEventListener("click", e => { const c = e.target.closest(".chip"); if (!c) return; if (c.dataset.mfxtext) { mfxPickText(); return; } mfxPick = { target: c.dataset.mfxtarget }; renderMfxChips(); mfxFetch(); }); const mfxGroupsChanged = () => { pref.setJson("wf-modstats-groups", mfxGroups()); if (mfxPick) mfxFetch(); }; $("#mfx-groups").addEventListener("change", e => { if (e.target.dataset.group) mfxGroupsChanged(); }); $("#mfx-all").addEventListener("click", () => { mfxBoxes().forEach(x => { x.checked = true; }); mfxGroupsChanged(); }); $("#mfx-none").addEventListener("click", () => { mfxBoxes().forEach(x => { x.checked = false; }); mfxGroupsChanged(); }); /* 開いたとき: データの有無だけ確かめる(結果と入力は残す。データ更新の後に開き直せば読み直す) */ loaders.modstats = async () => { applyViewText("modstats"); try { const res = await api("/api/modstats/suggest?q="); if (!res.has_data) { mfxCands = []; renderMfxChips(); mfxNote("modstats.nodata"); return; } } catch (e) { $("#mfx-out").innerHTML = noData(e.message); return; } if (mfxPick) await mfxFetch(); else mfxNote(mfxMsg === "modstats.nodata" ? "modstats.empty" : mfxMsg); }; relabel.modstats = () => { applyViewText("modstats"); renderMfxChips(); if (mfxRes) renderMfxResult(); else mfxNote(mfxMsg); };
