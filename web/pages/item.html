アイテムから探す

アイテム名(日本語・英語・ひらがな)を入力すると、ミッション・敵・バウンティ・レリックなどの入手先と確率を一覧にします。

/* ---------- アイテムから探す ---------- */ /* 素材・パーツ/設計図・MOD・レリック・その他(AREA_KINDS)を種類のチェックで選んで探す。エリアから探すの逆向き。 種類のチェックは localStorage(wf-item-kinds)に覚える。覚えていなければ ITEM_KIND_DEFAULT と同じ初期値。 カテゴリー(マスター一覧と同じ分け方)と「Prime を除外する」はパーツだけに効く(パーツにチェックがあるときだけ出す) */ const itemKinds = kindChecks("item", CONST.ITEM_KIND_DEFAULT, "wf-item-kinds"); const itemHasPart = () => itemKinds().includes("part"); const showItemPartOpt = () => { $("#item-partopt").hidden = !itemHasPart(); }; showItemPartOpt(); // パーツのカテゴリーと「Prime を除外する」(以前のパーツ検索で覚えたものを引き継ぐ) let itemCat = ""; itemCat = pref.get("wf-part-cat", ""); if (itemCat && !hasText("mastery.cat." + itemCat)) itemCat = ""; // 今は無いカテゴリーが残っていたら外す $("#item-noprime").checked = pref.flag("wf-part-noprime"); const itemSearch = sourceSearch({ id: "item", suggest: q => { const part = itemHasPart(); return `/api/suggest/item?q=${encodeURIComponent(q)}&kinds=${itemKinds().join(",")}` + `&cat=${encodeURIComponent(part ? itemCat : "")}&noprime=${part && $("#item-noprime").checked ? 1 : 0}`; }, list: res => res.items, tag: i => `${esc(t("area.kind." + i.kind))}`, // 種類を1つも選んでいないときは候補を取らない(空の kinds はサーバーで「指定なし = 既定」になるため) blocked: () => (itemKinds().length ? "" : t("item.nokind")), onSuggest: res => { if (!res || !itemHasPart()) { $("#item-cats").innerHTML = ""; return; } const all = res.cats.reduce((n, c) => n + c.n, 0); // 選んでいるカテゴリーに名前の合う候補が無いときも、ボタンは 0 件で残す(押し直して外せるように) const cats = itemCat && !res.cats.some(c => c.key === itemCat) ? [...res.cats, { key: itemCat, n: 0 }] : res.cats; $("#item-cats").innerHTML = `` + cats.map(c => ``).join(""); }, more: (res, items) => res.total > items.length ? `${esc(fmt("item.more", { n: res.total - items.length }))}` : "", msg: { noMatch: () => t("item.nomatch"), pick: () => t("item.pick"), empty: () => t("item.empty"), noData: () => t("item.nodata"), }, }); $("#item-kinds").addEventListener("change", e => { // 種類を変えたら(覚えるのは kindChecks)、候補を取り直す if (!e.target.dataset.kind) return; showItemPartOpt(); itemSearch.search(false); }); $("#item-cats").addEventListener("click", e => { const c = e.target.closest("[data-pcat]"); if (!c) return; itemCat = c.dataset.pcat; pref.set("wf-part-cat", itemCat); itemSearch.search(false); }); $("#item-noprime").addEventListener("change", e => { // 入手先がレリックのパーツ(Prime)を候補から外す。この PC に覚えさせる pref.setFlag("wf-part-noprime", e.target.checked); itemSearch.search(false); }); loaders.item = () => { applyViewText("item"); return itemSearch.load(); }; relabel.item = itemSearch.relabel; dataResets.item = () => itemSearch.reset();
