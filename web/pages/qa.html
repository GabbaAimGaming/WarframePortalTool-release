/* ---------- Q&A ---------- */ /* 答えの文は API(Python)が日本語で作る(Q&A は日本語のみ)。ここは表示とボタンだけ。枠の文字(ボタン・見出し)は I18N(t)から。 履歴は新しいものを上に QA_LOG_MAX 件まで(ページを離れても残す。データは取り直さない) */ const QA_LOG_MAX = 20; let qaLog = []; // [{res, done}] done = ビルド管理に追加した名前 {ビルド名: 保存した名前} async function qaAsk(params) { const qs = new URLSearchParams(params).toString(); try { const res = await api(`/api/qa?${qs}`); qaLog.unshift({ res, done: {} }); qaLog = qaLog.slice(0, QA_LOG_MAX); } catch (e) { qaLog.unshift({ err: e.message, q: params.q || "" }); } renderQa(); } // ボタン: o = {name, intent, choice, lore, only, menu, q}(どれも任意)。押すとその値で /api/qa に聞く。spoil はネタバレの確認を飛ばす。 // only は解説のその段だけ(「続き」。指定なしは 0段目からその段まで)。menu は「教えてクロちゃん」の分類、q は例文(その文で聞く)、 // talk はツールの使い方・雑談の話。page は聞かずに画面のページを開く(qaPageChip。無いページのボタンは出さない) const QA_CHIP_KEYS = ["name", "intent", "choice", "lore", "only", "menu", "q", "talk"]; const qaChip = (label, o, spoil) => ``; const qaPageChip = (label, page) => pageExists(page) ? `` : ""; const qaOption = o => o.page ? qaPageChip(o.label, o.page) : qaChip(o.label, o); // ツールの使い方・雑談の話のカード(文は API が日本語で返す) const qaTalkCard = talk => `
${esc(talk.title)}
${ talk.paragraphs.map(x => `

${esc(x)}

`).join("")}
`; // 解説のカード(文は API が日本語で返す)。見出しは 0段目(最初の段)のタイトル、2つ目からの段は小見出し。出典は wiki の記事の URL と、ファイル全体の credit・license function qaLoreCard(lore, next) { const body = lore.parts.map((p, i) => `${i ? `

${esc(p.title)}

` : ""}${ p.prefix ? `

${esc(p.prefix)}

` : ""}${p.paragraphs.map(x => `

${esc(x)}

`).join("")}`).join(""); const src = lore.source.map(u => `${esc(u.replace(/^https?:\/\//, ""))}`).join(" "); const credit = [lore.credit, lore.license].filter(Boolean).join(" / "); return `
${esc(lore.parts[0].title)}
${body}${credit || src ? `
${esc(t("qa.lore.source"))}: ${esc(credit)} ${src}
` : ""}
${ next ? `
${qaChip(next.label, { lore: next.id, only: "1" })}
` : ""}`; } function qaBuildCard(b, done) { const row = (key, val) => val ? `
${esc(t(key))}
${esc(val)}
` : ""; return `
${esc(b.name)}${esc(b.type_ja)} / ${esc(b.target_ja)} / ${esc(fmt("qa.b.forma", { n: b.forma }))}
${b.mods.length ? `
${esc(t("qa.b.mods"))}
    ${b.mods.map(m => `
  1. ${esc(m.name)}${m.effect ? `${esc(m.effect)}` : ""}
    2. `).join("")}
` : ""}${row("qa.b.arcanes", b.arcanes.join("、"))}${row("qa.b.helminth", b.helminth)}${ row("qa.b.shards", b.shards.join("\n"))}${row("qa.b.note", b.note)}
${done ? `${esc(fmt("qa.adopt.done", { name: done }))} ` : ``}
`; } // 時代ごとのレリックの入手先: その場所で出るレリックと確率(ドロップ表のまま) function qaRelics(relics) { if (!relics || !relics.length) return ""; return `

`; } function qaItem(entry, idx) { if (entry.err) return `
${esc(entry.q)}
${esc(entry.err)}
`; const r = entry.res; const q = r.q || (r.name ? r.name.ja : r.lore ? r.lore.title : r.spoiler ? r.spoiler.title : r.menu ? r.menu.title : r.talk ? r.talk.title : ""); let h = `

${r.facts.map(f => ``).join("")}
${esc(f.label)}${esc(f.value)}

${esc(g.title)}

`; } function renderQa() { $("#qa-jaonly").hidden = lang === "ja"; $("#qa-log").innerHTML = qaLog.length ? qaLog.map(qaItem).join("") : `

renderQa(); }); /* ビルド管理に追加: 名前を入れる小窓(初期値は「名前_dev」)。同じ名前があれば小窓のまま入れ直してもらう(上書きしない) */ let qaAdopt = null; // {idx, build} $("#qa-log").addEventListener("click", e => { const page = e.target.closest("[data-qapage]"); if (entry && entry.done) entry.done[qaAdopt.build] = res.name; $("#qa-dlg").close(); renderQa(); } catch (err) { $("#qa-dlg-err").textContent = err.message; } }); loaders.qa = () => { applyViewText("qa"); renderQa(); }; relabel.qa = () => { applyViewText("qa"); renderQa(); };
