要求素材集

Warframe・武器などの名前(日本語・英語・ひらがな)を入力すると、作るのに要る素材を部品ごとに表示します。

/* ---------- 要求素材集 ---------- */ /* 名前で候補(マスター一覧に載っていて設計図があるもの)を出し、選んだものの素材(/api/recipe)を入れ子の表にする。 部品(ニューロティック・シャーシなど)は下に開き、末端の素材には入手先を確率の高い順に RECIPE_SOURCE_LIMIT 件まで出す。 作り方が2通り以上あるもの(例: Brakk)は全部並べる。末端の素材の合計とクレジットの合計は作り方ごと。 候補と結果の取り方・言語の切り替えは parts.html の sourceSearch(結果の取り方と描き方だけこのページで渡す) */ const recipeNum = n => Number(n).toLocaleString(locale()); /* 作成時間(秒)→「1日 12時間」のように(秒は1分未満のときだけ) */ function recipeTime(sec) { const days = Math.floor(sec / 86400), hours = Math.floor(sec % 86400 / 3600), mins = Math.floor(sec % 3600 / 60); const out = []; if (days) out.push(fmt("recipe.time.d", { n: days })); if (hours) out.push(fmt("recipe.time.h", { n: hours })); if (mins) out.push(fmt("recipe.time.m", { n: mins })); return out.length ? out.join(" ") : fmt("recipe.time.s", { n: sec % 60 }); } const recipeCost = x => esc(fmt("recipe.cost", { cr: recipeNum(x.credits), time: recipeTime(x.time) })); /* 末端の素材の入手先(先頭の数件。残りは「ほか n か所」) */ function recipeSources(src) { if (!src || !src.rows.length) return `${esc(t("recipe.nosrc"))}`; const line = r => { const stage = r.stage === "—" ? "" : renderTxt(r.stage); // ローテーションの無い入手先は「—」なので出さない const where = [renderTxt(r.place), renderTxt(r.detail), stage].filter(Boolean).join(" / "); return `${esc(where)} ${r.chance == null ? "—" : esc(fmtPct(r.chance))}`; }; return src.rows.map(line).join("") + (src.total > src.rows.length ? `${esc(fmt("recipe.more", { n: src.total - src.rows.length }))}` : ""); } /* 入れ子の行(部品は下の行を続けて出す) */ function recipeRows(parts, depth) { return parts.map(p => { const name = `${esc(nameByLang(p))}`; const num = `${esc(fmt("recipe.count", { n: recipeNum(p.count) }))}`; if (p.kind === "part") { return `${name}${num}${recipeCost(p)}` + recipeRows(p.parts, depth + 1); } return `${name}${num}${recipeSources(p.sources)}`; }).join(""); } function recipeBlock(rec, idx, all) { const head = all > 1 ? `

${esc(fmt("recipe.variant", { i: idx + 1, n: all }))}

` : ""; const cols = ["name", "count", "cost", "src"].map(c => `${esc(t("recipe.col." + c))}`).join(""); const top = `${esc(nameByLang({ ja: rec.bp_ja, en: rec.bp_en }))}` + `${recipeCost(rec)}`; const totals = rec.total.items.map(x => `${esc(nameByLang(x))}${esc(recipeNum(x.count))}`).join(""); return `${head}
${cols}${top}${recipeRows(rec.parts, 1)}

${esc(t("recipe.total"))}

${totals}
${esc(t("recipe.col.item"))}${esc(t("recipe.col.sum"))}

${esc(fmt("recipe.credits", { cr: recipeNum(rec.total.credits) }))}

`; } const recipeSearch = sourceSearch({ id: "recipe", suggest: q => `/api/suggest/recipe?q=${encodeURIComponent(q)}`, list: res => res.items, tag: i => i.cat ? `${esc(tName("mastery.cat.", i.cat))}` : "", more: (res, items) => res.total > items.length ? `${esc(fmt("recipe.morecand", { n: res.total - items.length }))}` : "", result: key => `/api/recipe?key=${encodeURIComponent(key)}`, render: (r, nameOf, subOf) => `

${esc(nameOf(r))}

${subOf(r) ? `${esc(subOf(r))}` : ""}${r.cat ? `${esc(tName("mastery.cat.", r.cat))}` : ""}
${r.recipes.map((rec, i) => recipeBlock(rec, i, r.recipes.length)).join("")}

${esc(t("recipe.footnote"))}

`, msg: { noMatch: () => t("recipe.nomatch"), pick: () => t("recipe.pick"), empty: () => t("recipe.empty"), noData: () => t("recipe.nodata"), }, }); loaders.recipe = () => { applyViewText("recipe"); return recipeSearch.load(); }; relabel.recipe = recipeSearch.relabel; dataResets.recipe = () => recipeSearch.reset();
