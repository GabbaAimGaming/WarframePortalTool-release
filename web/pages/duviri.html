デュヴィリサーキット

サーキットで選べる報酬の今週分と、来週以降の並びです。毎週月曜 9:00(JST)のウィークリーリセットで切り替わります。

ゲーム内とズレているとき(手動で調整)
 
 

ゲーム内の今週の報酬に合う ID を選んで保存すると、以降はそこから毎週進めて表示します。調整はモードごとで、ツールの更新では消えません。「同梱の設定に戻す」で調整を消せます。

/* ---------- デュヴィリサーキット ---------- */ /* 今週の ID は API(Python)側で計算する。ここは表示・タブ・手動調整の送信だけ。文字はすべて I18N(t)から */ let duviri = null, dvMode = "normal"; const dvDate = iso => { const [, m, d] = iso.split("-"); return `${+m}/${+d}${t("duviri.weekday")}`; }; /* 報酬は英語名のまま表示する(Normal は Warframe、Steel Path は武器。JSON の綴りがそのまま出る) */ const dvItem = it => it ? esc(it) : ""; function dvTable(rows, cols, withDate) { const n = Array.from({ length: cols }, (_, i) => i); return `
${withDate ? `` : ""}${ n.map(i => ``).join("")}${rows.map(r => `${withDate ? `` : ""}${ n.map(i => ``).join("")}`).join("")}
${esc(t("duviri.date"))}${esc(t("duviri.col").replace("{n}", i + 1))}
${esc(dvDate(r.date))}${dvItem(r.items[i])}
`; } function renderDuviri() { applyViewText("duviri"); if (!duviri) return; const keys = duviri.modes.map(m => m.key); if (!keys.includes(dvMode)) dvMode = keys[0]; $("#dv-modes").innerHTML = duviri.modes.map(m => ``).join(""); const m = duviri.modes.find(x => x.key === dvMode); $("#dv-adjust").hidden = !m || !m.ok; if (!m || !m.ok) { $("#dv-out").innerHTML = noData(t("duviri.broken")); return; } $("#dv-out").innerHTML = `

${esc(t("duviri.current"))}${esc(fmt("duviri.from", { date: dvDate(m.current.date) }))}${ m.manual ? `${esc(t("duviri.manual"))}` : ""}

${dvTable([m.current], m.columns, false)}

${esc(t("duviri.upcoming"))}

${m.upcoming.length ? dvTable(m.upcoming, m.columns, true) : noData(t("duviri.none"))}`; $("#dv-id").innerHTML = m.rotation.map(r => ``).join(""); $("#dv-reset").disabled = !m.manual; } /* 選択中のタブを押しても何もしない(AC-1 / AC-3) */ $("#dv-modes").addEventListener("click", e => { const b = e.target.closest("[data-dvmode]"); if (!b || b.dataset.dvmode === dvMode) return; dvMode = b.dataset.dvmode; pref.set("wf-duviri-mode", dvMode); $("#dv-err").textContent = ""; renderDuviri(); }); async function dvPost(path, body) { $("#dv-err").textContent = ""; try { duviri = await api(path, { method: "POST", body }); renderDuviri(); } catch (e) { $("#dv-err").textContent = e.message; } } $("#dv-form").addEventListener("submit", e => { e.preventDefault(); dvPost("/api/duviri/adjust", { mode: dvMode, id: +$("#dv-id").value }); }); $("#dv-reset").addEventListener("click", () => dvPost("/api/duviri/reset", { mode: dvMode })); /* 開くたびに取り直す(週を跨いでも開き直せば新しくなる) */ async function loadDuviri() { dvMode = pref.get("wf-duviri-mode", dvMode); renderDuviri(); if (!duviri) $("#dv-out").innerHTML = `
${esc(t("duviri.loading"))}
`; try { duviri = await api("/api/duviri"); renderDuviri(); } catch (e) { $("#dv-modes").innerHTML = ""; $("#dv-adjust").hidden = true; $("#dv-out").innerHTML = noData(e.message); } } loaders.duviri = loadDuviri; relabel.duviri = renderDuviri;
