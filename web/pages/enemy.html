敵から探す

敵の名前(英語)を入力すると、その敵を1体倒したときに落とすアイテムと確率を表示します。

/* ---------- 敵から探す ---------- */ /* 敵の名前(英名)で候補を出し、選んだ敵が落とすアイテム(/api/enemy)を種類の順 → 確率の高い順に表にする。 種類のチェックは localStorage(wf-enemy-kinds)に覚える。覚えていなければ ENEMY_KIND_DEFAULT(すべて)。 入手先の表(sourceTable)の敵の名前を押すと、ここに移ってその敵を開く(openEnemy) */ const enemyKinds = kindChecks("enemy", CONST.ENEMY_KIND_DEFAULT, "wf-enemy-kinds"); let enemySel = null, enemyList = null, enemyRes = null; // 候補・結果は言語の切り替えで描き直すため手元に残す function renderEnemyChips() { const res = enemyList; if (!res) { $("#enemy-chips").innerHTML = ""; return; } $("#enemy-chips").innerHTML = res.items.length ? res.items.map(i => ``).join("") + (res.total > res.items.length ? `${esc(fmt("enemy.more", { n: res.total - res.items.length }))}` : "") : `${esc(t("enemy.nomatch"))}`; } const enemyChipsGuard = latestGuard(), enemyResGuard = latestGuard(); // 候補・結果の取得の追い越しの対策(parts.html) /* 候補を取り直す。返り値は候補の一覧(取得に失敗したとき・待っている間に次の取得が始まっていたときは null) */ async function loadEnemyChips() { const fresh = enemyChipsGuard(); const q = $("#enemy-q").value; if (!q.trim()) { enemyList = null; renderEnemyChips(); return []; } // 敵は千件以上あるので、何か入れるまで候補は出さない try { const res = await api(`/api/suggest/enemy?q=${encodeURIComponent(q)}`); if (!fresh()) return null; enemyList = res; renderEnemyChips(); return enemyList.items; } catch (e) { if (fresh()) { enemyList = null; enemyResGuard(); $("#enemy-chips").innerHTML = ""; $("#enemy-out").innerHTML = noData(e.message); } return null; } } function renderEnemyResult() { const r = enemyRes; if (!enemySel) { $("#enemy-out").innerHTML = `
${esc(t($("#enemy-q").value.trim() ? "enemy.pick" : "enemy.empty"))}
`; return; } if (!r) return; const kinds = r.kinds_n.map(k => `${esc(t("area.kind." + k.key))} ${k.n}`).join(""); const headHtml = `

${esc(r.name)}

${esc(fmt("enemy.count", { n: r.items.length, total: r.total }))}${kinds}
`; if (!r.items.length) { $("#enemy-out").innerHTML = headHtml + noData(t("enemy.none")); return; } const max = Math.max(...r.items.map(i => i.chance || 0), 0.0001); const head = ["item", "detail", "chance", "rarity"].map(c => `${esc(t("enemy.col." + c))}`).join(""); const rows = r.items.map(i => `${esc(nameByLang(i))}${esc(t("area.kind." + i.kind))} ${esc(renderTxt(i.detail))}
${i.qty ? esc(fmt("src.qty", { n: i.qty })) : ""}${fmtPct(i.chance)}
${i.rarity ? rarity(i.rarity) : ""}`).join(""); $("#enemy-out").innerHTML = headHtml + `
${head}${rows}

${esc(t("enemy.footnote"))}

`; } async function loadEnemyResult() { const fresh = enemyResGuard(); enemyRes = null; if (!enemySel) { renderEnemyResult(); return; } const kinds = enemyKinds(); if (!kinds.length) { $("#enemy-out").innerHTML = `
${esc(t("enemy.nokind"))}
`; return; } try { const res = await api(`/api/enemy?name=${encodeURIComponent(enemySel)}&kinds=${kinds.join(",")}`); if (!fresh()) return; enemyRes = res; renderEnemyResult(); } catch (e) { if (fresh()) $("#enemy-out").innerHTML = noData(e.message); } } async function searchEnemy() { const items = await loadEnemyChips(); if (items === null) return; // 取得に失敗(エラーは表示済み)か、次の取得が始まっている const q = $("#enemy-q").value.trim().toLowerCase(); const exact = q && items.find(i => i.name.toLowerCase() === q); if (exact || items.length === 1) enemySel = (exact || items[0]).key; else if (!items.some(i => i.key === enemySel)) enemySel = null; // 選んでいた敵が候補から外れたら選び直し renderEnemyChips(); await loadEnemyResult(); } /** 敵から探すでその敵を開く(入手先の表の敵の名前から) */ function openEnemy(name) { $("#enemy-q").value = name; enemySel = name; show("enemy"); scrollView(); } $("#enemy-q").addEventListener("input", debounce(searchEnemy)); $("#enemy-kinds").addEventListener("change", e => { // 種類を変えたら(覚えるのは kindChecks)、表を取り直す if (!e.target.dataset.kind) return; loadEnemyResult(); }); $("#enemy-chips").addEventListener("click", e => { const c = e.target.closest(".chip[data-key]"); if (!c) return; enemySel = c.dataset.key; renderEnemyChips(); loadEnemyResult(); }); document.addEventListener("click", e => { const b = e.target.closest("[data-enemy]"); if (b) openEnemy(b.dataset.enemy); }); loaders.enemy = async () => { applyViewText("enemy"); const s = await refreshStatus(); if (!s || !s.has_drops) { $("#enemy-chips").innerHTML = ""; $("#enemy-out").innerHTML = noData(t("enemy.nodata")); return; } await loadEnemyChips(); await loadEnemyResult(); }; relabel.enemy = () => { applyViewText("enemy"); renderEnemyChips(); renderEnemyResult(); };
