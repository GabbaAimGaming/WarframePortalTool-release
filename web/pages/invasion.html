侵略サイクル

いま進行中の侵略と、侵略で手に入る武器パーツの取得状況です。侵略の一覧は「公式データで更新」を押したときに取り直します。

侵略ミッション

獲得パーツ一覧

報酬はその陣営の側でミッションを3回クリアすると手に入ります。完了した侵略は表示しません。取得済みのチェックはこの PC に保存され、ツールの更新では消えません。

/* ---------- 侵略サイクル ---------- */ /* 一覧・照合は API(Python)側で作る。ここは表示とチェックの送信だけ。文字はすべて I18N(t)から。 侵略の一覧を取り直すのは「公式データで更新」を押したときだけ(開いたときは保存済みの結果を出す) */ let inv = null; const invFmt = (key, vals) => Object.entries(vals).reduce((s, [k, v]) => s.split("{" + k + "}").join(v), t(key)); /* 陣営名: 表に無い値(新しい陣営)は World State の値のまま出す */ const invFaction = f => { const k = "invasion.faction." + f; return I18N.ja[k] != null ? t(k) : f; }; const invFac = f => `${esc(invFaction(f))}`; const invPart = (weapon, kind) => `${weapon} ${t("invasion.part." + kind)}`; function invPlace(r) { const ja = lang === "ja", node = ja ? r.node_ja : r.node_name, planet = ja ? r.planet_ja : r.planet; return planet ? invFmt("invasion.node", { node, planet }) : node; } function invReward(r) { const cnt = r.count > 1 ? invFmt("invasion.count", { n: r.count }) : ""; if (r.part) return `${esc(invPart(r.part.weapon, r.part.kind))}${esc(cnt)}${ r.part.owned ? `${esc(t("invasion.owned"))}` : ""}`; return `${esc(nameByLang(r, "name_ja", "name"))}${esc(cnt)}`; } function invSide(faction, rewards) { return `
${invFmt("invasion.side", { f: invFac(faction) })}${ rewards.length ? rewards.map(r => `
${invReward(r)}
`).join("") : `
${esc(t("invasion.noreward"))}
`}
`; } function invBar(r) { if (r.progress == null) return ""; const p = r.progress * 100; if (r.infested) return `
${esc(invFmt("invasion.progress", { p: p.toFixed(1) }))}
`; return `
${esc(invFaction(r.attacker))} ${p.toFixed(1)}%${esc(invFaction(r.defender))} ${(100 - p).toFixed(1)}%
`; } function invCard(r) { /* 感染体が攻め側のときは攻め側の報酬が無い(空)ので、守り側だけを出す */ const sides = (r.infested && !r.attacker_reward.length ? "" : invSide(r.attacker, r.attacker_reward)) + invSide(r.defender, r.defender_reward); return `
${esc(invPlace(r))}${ invFmt("invasion.vs", { a: invFac(r.attacker), d: invFac(r.defender) })}
${invBar(r)}
${sides}
`; } function renderInvasionList() { if (!inv.has_snapshot) return noData(t("invasion.nodata")); if (inv.needs_sync) return noData(t("invasion.stale")); if (!inv.invasions.length) return noData(t("invasion.none")); return (inv.regions_ok ? "" : `

${esc(t("invasion.noregions"))}

`) + inv.invasions.map(invCard).join(""); } function renderInvasionParts() { if (!inv.parts_ok) return noData(t("invasion.broken")); return inv.weapons.map(w => `
${esc(w.name)}${w.owned} / ${w.total}
${w.parts.map(p => ``).join("")}
`).join(""); } function renderInvasion() { applyViewText("invasion"); if (!inv) return; $("#inv-list").innerHTML = renderInvasionList(); $("#inv-parts").innerHTML = renderInvasionParts(); const owned = inv.weapons.reduce((n, w) => n + w.owned, 0), total = inv.weapons.reduce((n, w) => n + w.total, 0); $("#inv-sum").textContent = inv.parts_ok ? invFmt("invasion.sum", { owned, total }) : ""; $("#inv-unknown").innerHTML = inv.unknown_parts.length ? `
${esc(t("invasion.unknown"))}
${inv.unknown_parts.map(u => `
${esc(u.name)} ${esc(u.path)}
`).join("")} ${esc(t("invasion.unknown.help"))}
` : ""; $("#inv-info").textContent = inv.fetched_at ? invFmt("invasion.fetched", { at: new Date(inv.fetched_at * 1000).toLocaleString( lang === "ja" ? "ja-JP" : "en-US", { month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit" }) }) : ""; } $("#inv-sync").addEventListener("click", async () => { const btn = $("#inv-sync"); btn.disabled = true; $("#inv-err").textContent = ""; $("#inv-info").textContent = t("invasion.syncing"); try { inv = await api("/api/invasions/update", { method: "POST", body: {} }); renderInvasion(); if (inv.synced && inv.synced.cached) $("#inv-info").textContent += t("invasion.cached"); } catch (e) { $("#inv-err").textContent = e.message; renderInvasion(); } btn.disabled = false; }); /* チェックの付け外し: 返ってきた全体で描き直す(左の「取得済み」の表示もそろう)。失敗したらチェックを戻す */ $("#inv-parts").addEventListener("change", async e => { const box = e.target.closest("input[data-path]"); if (!box) return; $("#inv-err").textContent = ""; try { inv = await api("/api/invasions/parts", { method: "POST", body: { path: box.dataset.path, owned: box.checked } }); renderInvasion(); } catch (err) { box.checked = !box.checked; $("#inv-err").textContent = err.message; } }); async function loadInvasion() { renderInvasion(); if (!inv) $("#inv-list").innerHTML = `
${esc(t("invasion.loading"))}
`; try { inv = await api("/api/invasions"); renderInvasion(); } catch (e) { $("#inv-list").innerHTML = noData(e.message); } } relabel.invasion = renderInvasion; loaders.invasion = loadInvasion; // 開くたびに読み直す(ほかのページの「時刻合わせ」で取り直した結果も反映)
