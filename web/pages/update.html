データ更新

ドロップ表・敵やバウンティの入手先・Riven 相場・日本語名の辞書(Riven性質・シミュレータ用のデータを含む)を取得します。ゲームのアップデート後にも実行してください。

calamity-inc/warframe-public-export-plus(ゲームのデータから作られた非公式のデータ)から取ります。設定は取得したときに覚え、次からも同じにします(コマンドの update も同じ設定に従います)。オフにして取得すると、前回取った分は消えます。

入手先に出す商人

チェックした商人の品物を入手先に出します(すぐ反映されます)。「名前未登録」はゲームのアップデートで増えた商人などで、名前はデータから作っています。

/* ---------- ツールの手動の更新(/api/app-update/*)----------
   ① 確認中 → ② 最新なら知らせて OK / ③ 新しい版があれば「更新しますか」(再起動の警告つき)→ ④ いいえなら閉じる /
   ⑤ はいなら更新中(ボタンなし)→ ⑥ 完了(OK で起動し直す。bat から起動していなければ手で起動し直す案内)。小窓の文字は開くときの言語 */
const APP_RESTART_FIRST = 1500, APP_RESTART_POLL = 1000, APP_RESTART_WAIT = 60000;  // 起動し直すのを待つ: 最初の待ち・聞き直す間隔・上限(ミリ秒)
const appUpdOk = () => [{ label: t("dlg.ok"), value: "ok" }];
const appUpdSay = (text, cls) => appModal.show({ lines: [{ text, cls }], buttons: appUpdOk() });
const appUpdState = r => fmt("appupd.state." + r.state, { current: r.current || "", remote: r.remote || "" });
const appUpdWait = ms => new Promise(res => setTimeout(res, ms));
async function appUpdCall(path) {
  try { return await api(path, { method: "POST", body: {} }); }
  catch (e) { appUpdSay(e.message); return null; }
}
async function checkAppUpdate() {
  appModal.show({ lines: [{ text: t("appupd.checking") }] });
  const r = await appUpdCall("/api/app-update/check");
  if (!r) return;
  if (r.state !== "newer") { appUpdSay(appUpdState(r)); return; }
  const go = await appModal.show({
    lines: [{ text: fmt("appupd.ask", { remote: r.remote }) }, { text: t("appupd.warn." + r.restart), cls: "warn" }],
    buttons: [{ label: t("dlg.yes"), value: "yes" }, { label: t("dlg.no"), value: "", cls: "ghost" }],
  });
  if (go === "yes") await applyAppUpdate();
}
async function applyAppUpdate() {
  appModal.show({ lines: [{ text: t("appupd.applying") }] });
  const r = await appUpdCall("/api/app-update/apply");
  if (!r) return;
  if (r.state !== "updated") { appUpdSay(r.state === "failed" ? t("appupd.apply.failed") : appUpdState(r)); return; }
  await appModal.show({ lines: [{ text: t("appupd.done") }, { text: t("appupd.done." + r.restart) }], buttons: appUpdOk() });
  if (r.restart === "auto") await restartApp();
}
/* 起動し直してもらい、サーバーが戻ったらページを読み直す(新しい画面にする)。戻らなければ手で起動し直す案内 */
async function restartApp() {
  appModal.show({ lines: [{ text: t("appupd.restarting") }] });
  if (!await appUpdCall("/api/app-update/restart")) return;
  await appUpdWait(APP_RESTART_FIRST);  // 止まるまで(止まる前の応答を「戻った」と見ない)
  for (const until = Date.now() + APP_RESTART_WAIT; Date.now() < until; await appUpdWait(APP_RESTART_POLL)) {
    try { await api("/api/status"); location.reload(); return; } catch (e) { /* まだ起動していない */ }
  }
  appUpdSay(t("appupd.restart.timeout"));
}

function renderOcrPanel() {
  const box = $("#ocr-panel");
  box.hidden = !betaOn;
  if (!ocrPanel || box.hidden) return;
  const info = ocrPanel, rt = info.runtime || {};
  const row = (k, v) => `

`;
  const using = info.available ? tName("ocr.engine.name.", info.engine) + (info.fallback ? t("ocr.engine.fallback") : "") : ocrReason(info);
  const parts = rt.ready ? t("ocr.engine.rt.ready") : !rt.supported && !rt.modules ? fmt("ocr.engine.rt.pyver", { tag: rt.tag || "" })
    : t("ocr.engine.rt.none");
  $("#ocr-info").innerHTML = row("ocr.engine.using", using) + row("ocr.engine.rt", parts);
  const options = (sel, values, prefix, current) => {
    sel.innerHTML = values.map(v => ``).join("");
  };
  options($("#ocr-engine"), info.choices.engine, "ocr.engine.name.", info.prefs.engine);
  options($("#ocr-tier"), info.choices.tier, "ocr.engine.tiername.", info.prefs.tier);
  $("#ocr-install").disabled = !!rt.ready || (!rt.supported && !rt.modules) || ocrJobRunning();
  renderOcrJob();
}
/* 部品の導入の進み具合(/api/ocr/runtime/progress)。導入は裏で動くので、取得中は毎秒聞き直す(everySecond.update) */
let ocrJob = null, ocrPolling = false;
const ocrJobRunning = () => !!ocrJob && ocrJob.state === "running";
const mbText = bytes => (bytes / 1048576).toLocaleString(locale(), { minimumFractionDigits: 1, maximumFractionDigits: 1 });
function renderOcrJob() {
  const box = $("#ocr-progress");
  box.hidden = !ocrJobRunning();
  if (box.hidden) return;
  const job = ocrJob;
  const step = job.phase === "fetch" || !job.phase ? "" : ` ${t("ocr.engine.phase." + job.phase)}`;
  const file = job.count ? fmt("ocr.engine.progress.file", { index: job.index, count: job.count, name: job.name || "" }) : t("ocr.engine.progress.prepare");
  box.innerHTML = `${progressBar(ratePct(job.done, job.total))}
    ${esc(file)}${esc(step)}
    ${esc(fmt("ocr.engine.progress.size", { done: mbText(job.done), total: mbText(job.total), time: dur(job.elapsed) }))}`;
}
async function ocrJobFinished(job) {
  const log = $("#ocr-log");
  if (job.state === "done") { log.hidden = false; showLog(log, { lines: job.lines, ok: true }); }
  else $("#ocr-err").textContent = job.error ? renderTxt(job.error) : "";
  forgetOcrInfo();
  await loadOcrPanel();
}
async function pollOcrJob() {
  if (ocrPolling) return;
  ocrPolling = true;
  try {
    const was = ocrJobRunning(), job = await api("/api/ocr/runtime/progress");
    ocrJob = job;
    if (was && !ocrJobRunning()) await ocrJobFinished(job);
    else renderOcrPanel();
  } catch (e) { $("#ocr-err").textContent = e.message; }
  finally { ocrPolling = false; }
}
everySecond.update = () => { if (ocrJobRunning()) pollOcrJob(); };
async function loadOcrPanel() {
  if (!betaOn) { renderOcrPanel(); return; }
  const fresh = ocrPanelGuard();
  try { const r = await api("/api/ocr-info"); if (!fresh()) return; ocrPanel = r; $("#ocr-err").textContent = ""; }
  catch (e) { if (fresh()) $("#ocr-err").textContent = e.message; }
  renderOcrPanel();
}
async function saveOcrPrefs(body) {
  const fresh = ocrPanelGuard();
  $("#ocr-err").textContent = "";
  try { const r = await api("/api/ocr/prefs", { method: "POST", body }); if (!fresh()) return; ocrPanel = r; forgetOcrInfo(); }
  catch (e) { if (fresh()) $("#ocr-err").textContent = e.message; }
  renderOcrPanel();
}
$("#ocr-engine").addEventListener("change", e => saveOcrPrefs({ engine: e.target.value }));

/* 入手先に出す商人(Export Plus の商人。選択は利用者データに保存し、入手先の検索・MOD・マスター一覧・Q&A・CLI に効く) */
let updVendors = null;  // GET /api/offers/vendors の vendors(読み込む前は null)
function loadUpdVendors() {
  return loadInto($("#update-vendors"), () => api("/api/offers/vendors"), r => { updVendors = r.vendors; renderUpdVendors(); });
}
function renderUpdVendors() {
  const box = $("#update-vendors");
  if (!updVendors) return;
  if (!updVendors.length) { box.innerHTML = `
${esc(t("upd.vendors.none"))}
`; return; }
  box.innerHTML = updVendors.map(v => {
    const name = renderTxt(v.name);
    return ``;
  }).join("");
}
$("#update-vendors").addEventListener("change", async e => {
  const input = e.target.closest("input[data-key]");
  const v = input && (updVendors || []).find(x => x.key === input.dataset.key);
  if (!v) return;
  const on = input.checked;
  // 描き直さずにチェックだけ合わせる(キーボードの位置を保つ)
  const ok = await optimistic({
    apply: () => { v.on = on; }, undo: () => { v.on = !on; }, render: () => { input.checked = v.on; },
    send: () => api("/api/offers/vendors", { method: "POST", body: { key: v.key, on } }),
    err_box: $("#update-vendors-err"),
  });
  if (ok) Object.values(dataResets).forEach(reset => reset());  // 入手先が変わったので、各ページの手元のデータを捨てる
});
