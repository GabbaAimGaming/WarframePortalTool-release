Prime の入手時期

いま入手できる Prime を、出てから長い順に並べています。

フレームの実装順

※ Prime はあるが入手不可
/* ---------- Prime の入手時期 ---------- */ let prData = null; /* 色: 並び順の段階(level 1 = 赤 = 次に Vault 入り、0 = 緑 = 最新)。赤 → 橙 → 黄 → 緑 の色相 */ const prColor = level => `hsl(${Math.round(120 * (1 - level))}, 70%, 48%)`; const prMonth = month => { const [year, mon] = month.split("-").map(Number); return new Date(year, mon - 1, 1).toLocaleDateString(locale(), { year: "numeric", month: "2-digit" }); }; const prDay = sec => new Date(sec * 1000).toLocaleDateString(locale(), { year: "numeric", month: "2-digit", day: "2-digit" }); const prName = item => item ? `${esc(nameByLang(item))}` : ""; /* 名前の3列(武器が3つ以上なら武器2の欄に「 / 」でつなぐ) */ function prNameCells(rel) { const frame = rel.frames.map(prName).join(" / "), wpn_one = prName(rel.weapons[0]); const wpn_two = rel.weapons.slice(1).map(prName).join(" / "); return `
${frame}
${wpn_one}
${wpn_two}
`; } function prGaugeRow(rel) { const width = rel.blocks / prData.months * 100; // 今月から出た月までのマス const label = fmt("primes.months", { month: prMonth(rel.month), n: rel.months }); const tag = rel.state === "pending" ? `${esc(t("primes.state.pending"))}` : ""; return prNameCells(rel) + `
` + `
${tag}
`; } function prHeadRow() { const cols = ["frame", "weapon1", "weapon2"].map(c => `
${esc(t("primes.col." + c))}
`).join(""); const scale = ["now", "year1", "year2"].map(c => `${esc(t("primes.scale." + c))}`).join(""); return cols + `
${scale}
`; } /* Prime Resurgence: 横いっぱいの青のゲージに期間。期間の外・対象なしは「期間外」 */ function prResRows(res) { const period = res.active && res.activation && res.expiry ? fmt("primes.res.period", { start: prDay(res.activation), end: prDay(res.expiry) }) : ""; const bar = `
${esc(period || t("primes.res.off"))}
`; const rows = res.rows.length ? res.rows.map(rel => prNameCells(rel) + bar).join("") : `
${bar}`; return `
${esc(t("primes.res.title"))}
` + rows; } function prNotes() { const notes = []; if (!prData.ready) notes.push(t("primes.unchecked")); if (prData.unknown.length) notes.push(fmt("primes.unknown", { names: prData.unknown.join(", ") })); return notes.map(n => `

${esc(n)}

`).join(""); } function renderPrimes() { applyViewText("primes"); if (!prData) return; const gauge = prData.gauge.length ? prData.gauge.map(prGaugeRow).join("") : `
${noData(t("primes.empty"))}
`; const as_of = prData.as_of ? `

${esc(fmt("primes.asof", { date: prData.as_of }))}

` : ""; $("#pr-out").innerHTML = prNotes() + `
${prHeadRow()}${gauge}${prResRows(prData.resurgence)}
` + as_of; renderPrimeOrder(); } /* 実装順の小窓の中身(記号は I18N。title と aria-label に文) */ function renderPrimeOrder() { if (!prData) return; const head = ["no", "frame", "prime"].map(c => `${esc(t("primes.order.col." + c))}`).join(""); const body = prData.order.map(row => { const label = t(`primes.mark.${row.mark}.label`); const mark = hasText("primes.mark." + row.mark) ? t("primes.mark." + row.mark) : ""; // 未実装(none)は記号なし return `${row.no}${prName(row.frame)}` + `${esc(mark)}`; }).join(""); $("#pr-order").innerHTML = `${head}${body}
`; } function loadPrimes() { applyViewText("primes"); $("#pr-err").textContent = ""; return loadInto($("#pr-out"), () => api("/api/primes"), data => { prData = data; renderPrimes(); }); } /* 小窓はヘッダーの下に出す(モーダルにすると幕がヘッダーまで覆うので、自前の幕 #pr-veil と show() で開く) */ function placePrimeOrder() { const head = $(".app > header") || $("header"); const top = Math.max(0, head ? head.getBoundingClientRect().bottom : 0); $("#pr-veil").style.top = `${top}px`; $("#pr-dlg").style.top = `${top + 16}px`; $("#pr-dlg").style.maxHeight = `calc(100dvh - ${top + 32}px)`; } function openPrimeOrder() { if (!prData) { $("#pr-err").textContent = t("common.loading"); return; } placePrimeOrder(); $("#pr-veil").hidden = false; $("#pr-dlg").show(); $("#pr-dlg-close").focus(); } const closePrimeOrder = () => { if ($("#pr-dlg").open) $("#pr-dlg").close(); }; $("#pr-dlg").addEventListener("close", () => { $("#pr-veil").hidden = true; $("#pr-order-open").focus(); }); $("#pr-order-open").addEventListener("click", openPrimeOrder); $("#pr-dlg-close").addEventListener("click", closePrimeOrder); $("#pr-veil").addEventListener("click", closePrimeOrder); document.addEventListener("keydown", e => { if (e.key === "Escape" && $("#pr-dlg").open) closePrimeOrder(); }); window.addEventListener("resize", () => { if ($("#pr-dlg").open) placePrimeOrder(); }); relabel.primes = renderPrimes; loaders.primes = loadPrimes; // 開くたびに読み直す(データ更新で取り直した結果も反映)
