Riven相場

公式が毎週公開している PC 版の Riven MOD 取引統計です。単位はプラチナ。


        

        

        
極端な出品価格が混ざるため、平均より中央値が目安になります。件数が少ない武器は振れ幅が大きくなります。

      


/* ---------- Riven 相場 ---------- */
let rivens = null;
const rivSort = { key: "um", dir: "desc" };
const rivGetters = {
  w: r => r.veiled ? "" : rivWeapon(r).toLowerCase(), t: r => rivType(r.type),
  um: r => r.unrolled ? r.unrolled.median : null, ua: r => r.unrolled ? r.unrolled.avg : null, up: r => r.unrolled ? r.unrolled.pop : null,
  rm: r => r.rerolled ? r.rerolled.median : null, ra: r => r.rerolled ? r.rerolled.avg : null, rp: r => r.rerolled ? r.rerolled.pop : null,
};
const plat = v => Math.round(v).toLocaleString(locale());
/* 武器種(I18N の riven.type.<英語名>。表に無ければ " Riven Mod" を除いた英語名)・武器名(日本語の画面は日本語名) */
const rivType = k => (hasText("riven.type." + k) ? t("riven.type." + k) : k.replace(" Riven Mod", ""));
const rivWeapon = r => (r.veiled ? t("riv.veiled") : nameByLang(r, "weapon_ja", "weapon_en"));
function rivCells(st) {
  if (!st) return `————`;
  return `${plat(st.median)}${plat(st.avg)}${esc(fmt("common.range", { a: plat(st.min), b: plat(st.max) }))}${Math.round(st.pop)}`;
}
function renderRivens() {
  if (!rivens) return;
  const q = $("#riv-q").value.trim().toLowerCase(), ty = $("#riv-type").value, veiled = $("#riv-veiled").checked;
  const list = sortRows(rivens.rows.filter(r => (!ty || r.type === ty) && (veiled || !r.veiled) &&
    (!q || [r.weapon_ja, r.weapon_en, rivType(r.type)].some(x => (x || "").toLowerCase().includes(q)))), rivSort, rivGetters);
  $("#riv-summary").innerHTML = `${fmt("riv.summary", { shown: `${list.length}`, total: rivens.rows.length })}${esc(fmt("riv.updated", { at: rivens.updated }))}`;
  $("#riv-out").innerHTML = list.length ? `
${["u", "r"].map(p => `${th(esc(t("riv.median")), p + "m", rivSort, "num")}${th(esc(t("riv.avg")), p + "a", rivSort, "num")}${th(esc(t("riv.pop")), p + "p", rivSort, "num")}`).join("")}${list.map(r => `${rivCells(r.unrolled)}${rivCells(r.rerolled)}`).join("")}

    
${esc(t("riv.col.weapon"))}
      ${esc(t("riv.col.type"))}
      ${esc(t("riv.unrolled"))}${esc(t("riv.rerolled"))}

      
${esc(t("riv.range"))}

    
${esc(rivWeapon(r))}${r.veiled ? "" : subEn(r, "weapon_ja", "weapon_en")}
      ${esc(rivType(r.type))}
`
    : noData(t("riv.none"));
}
onSortClick($("#riv-out"), rivSort, renderRivens, ["um", "ua", "up", "rm", "ra", "rp"]);
$("#riv-q").addEventListener("input", debounce(renderRivens, 150));
$("#riv-type").addEventListener("change", renderRivens);
$("#riv-veiled").addEventListener("change", renderRivens);
/* 武器種の選択肢(言語を切り替えたら作り直す。選んでいる値は残す) */
function renderRivenTypes() {
  const keep = $("#riv-type").value, types = [...new Set(rivens.rows.map(r => r.type))];
  $("#riv-type").innerHTML = `` + types.map(k => ``).join("");
  $("#riv-type").value = keep;
}
loaders.riven = async () => {
  applyViewText("riven");
  if (rivens) { renderRivenTypes(); renderRivens(); return; }
  try {
    rivens = await api("/api/rivens");
    renderRivenTypes(); renderRivens();
  } catch (e) { $("#riv-summary").innerHTML = ""; $("#riv-out").innerHTML = noData(e.message); }
};
relabel.riven = () => { applyViewText("riven"); relabelLogs(); if (rivens) { renderRivenTypes(); renderRivens(); } };
dataResets.riven = () => { rivens = null; };

/* ---------- Riven 相場だけ更新 ---------- */
async function updateRivensOnly(btn, log) {
  btn.disabled = true; const label = btn.textContent; btn.textContent = t("riv.fetching");
  log.hidden = false; log.textContent = t("riv.fetching.log");
  try {
    const r = await api("/api/update-rivens", { method: "POST", body: {} });
    showLog(log, r);
    if (r.ok) { rivens = null; if (!$("#view-riven").hidden) await loaders.riven(); }
  } catch (e) { log.textContent = e.message; }
  btn.disabled = false; btn.textContent = label; refreshStatus();
}
$("#riv-update").addEventListener("click", () => updateRivensOnly($("#riv-update"), $("#riv-log")));
$("#update-riv-btn").addEventListener("click", () => updateRivensOnly($("#update-riv-btn"), $("#update-log")));
